Private Sub Workbook_Open()
    On Error GoTo ErrorHandler ' Habilita o tratamento de erros
    
    ' Declaração de variáveis
    Dim shell As Object
    Dim printerName As String
    Dim ipAddress As String
    Dim configFile As String
    Dim fileNumber As Integer
    Dim ws As Worksheet
    
    ' Define a worksheet onde estão os dados
    Set ws = ThisWorkbook.Worksheets("QR")
    
    ' Obtém o nome da impressora e o IP das células da planilha
    printerName = Trim(ws.Range("B10").Value)
    ipAddress = Trim(ws.Range("B11").Value)
    
    ' Verifica se os valores foram preenchidos
    If printerName = "" Then
        MsgBox "O nome da impressora não foi definido na célula B10.", vbExclamation, "Erro de Configuração"
        Exit Sub
    End If
    If ipAddress = "" Then
        MsgBox "O endereço IP não foi definido na célula B11.", vbExclamation, "Erro de Configuração"
        Exit Sub
    End If
    
    ' Caminho temporário para salvar o arquivo de configuração
    configFile = Environ("TEMP") & "\ZebraConfigFile.txt"
    
    ' Remove qualquer arquivo de configuração anterior
    If Dir(configFile) <> "" Then Kill configFile
    
    ' Criação do arquivo de configuração
    fileNumber = FreeFile
    Open configFile For Output As #fileNumber
    
    ' Escreve o conteúdo do arquivo diretamente, linha por linha
    Print #fileNumber, "[FileInfo]"
    Print #fileNumber, "Type=Driver Configuration File"
    Print #fileNumber, "Printer Type=201"
    Print #fileNumber, "[Barcodes]"
    Print #fileNumber, "BarcodeNum=4"
    Print #fileNumber, "[BARFONT0]"
    Print #fileNumber, "Name=EAN 13"
    Print #fileNumber, "UseHeight=1"
    Print #fileNumber, "JoinMode=2"
    Print #fileNumber, "StripTrailingSpaces=1"
    Print #fileNumber, "Kind=0"
    Print #fileNumber, "ID=1"
    Print #fileNumber, "Height=160"
    Print #fileNumber, "Expansion=1"
    Print #fileNumber, "Ratio=16908546"
    Print #fileNumber, "Flags=6164"
    Print #fileNumber, "[BARFONT1]"
    Print #fileNumber, "Name=Code128"
    Print #fileNumber, "UseHeight=1"
    Print #fileNumber, "JoinMode=2"
    Print #fileNumber, "StripTrailingSpaces=1"
    Print #fileNumber, "Kind=0"
    Print #fileNumber, "ID=11"
    Print #fileNumber, "Height=160"
    Print #fileNumber, "Expansion=1"
    Print #fileNumber, "Ratio=16974083"
    Print #fileNumber, "Flags=20"
    Print #fileNumber, "[Parameters]"
    Print #fileNumber, "PaperSize=256"
    Print #fileNumber, "FormName=U"
    Print #fileNumber, "PageWidth=1000"
    Print #fileNumber, "PageHeight=340"
    Print #fileNumber, "HDPI=203"
    Print #fileNumber, "VDPI=203"
    Print #fileNumber, "Orientation=0"
    Print #fileNumber, "Quantity=1"
    Print #fileNumber, "Darkness=15"
    Print #fileNumber, "Speed=5"
    Print #fileNumber, "SensorType=0"
    Print #fileNumber, "RibbonSaver=0"
    Print #fileNumber, "MediaType=1"
    Print #fileNumber, "PauseLabel=0"
    Close #fileNumber
    
    ' Configurações da impressora
    Set shell = CreateObject("WScript.Shell")
    Dim command As String
    command = "cmd /c ""ZebraSetupUtility.exe -loadconfig=""" & configFile & """ -printer=""" & printerName & """ -ip=""" & ipAddress & """"
    shell.Run command, 0, True
    
    ' Mensagem de sucesso
    MsgBox "Configurações da impressora Zebra aplicadas com sucesso.", vbInformation, "Configuração Zebra"
    Exit Sub

ErrorHandler:
    ' Trata erros e exibe mensagens amigáveis
    MsgBox "Erro ao configurar a impressora Zebra." & vbCrLf & _
           "Detalhes: " & Err.Description, vbCritical, "Erro"
End Sub

Sub GerarQRCodeESequencia()
    Dim wsQR As Worksheet
    Dim wsAux As Worksheet
    Dim qtdeEtiquetas As Integer
    Dim i As Integer
    Dim urlBase As String
    Dim textoQR As String
    Dim QRImageURL As String
    Dim http As Object
    Dim img As Picture
    Dim cmToPoints As Double
    
    ' Conversão de cm para pontos (1 cm = 28,35 pontos)
    cmToPoints = 28.35
    
    ' Definir as planilhas
    Set wsQR = ThisWorkbook.Sheets("QR")
    Set wsAux = ThisWorkbook.Sheets("ABA AUXILIAR")
    
    ' Captura a quantidade de etiquetas da célula B11
    qtdeEtiquetas = wsQR.Range("D5").Value
    
    ' Verifica se B11 é um número válido maior que zero
    If IsNumeric(qtdeEtiquetas) And qtdeEtiquetas > 0 Then
        ' Loop para gerar a sequência, QR Code e imprimir
        For i = 1 To qtdeEtiquetas
            ' Escreve o número na célula B10
            wsQR.Range("D7").Value = i
            
            ' Captura o texto da célula 'ABA AUXILIAR'!$B$7
            textoQR = wsAux.Range("B7").Value
            
            ' Cria a URL do QR Code com base no texto
            urlBase = "https://quickchart.io/qr?text="
            QRImageURL = urlBase & Application.EncodeURL(textoQR)
            
            ' Baixa a imagem do QR Code
            Set http = CreateObject("MSXML2.XMLHTTP")
            http.Open "GET", QRImageURL, False
            http.Send
            
            ' Salva a imagem em um arquivo temporário
            If http.Status = 200 Then
                Dim TempFilePath As String
                TempFilePath = Environ("TEMP") & "\QRCode" & i & ".png"
                Dim stream As Object
                Set stream = CreateObject("ADODB.Stream")
                stream.Type = 1 ' Tipo binário
                stream.Open
                stream.Write http.responseBody
                stream.SaveToFile TempFilePath, 2
                stream.Close
                
                ' Insere a imagem do QR Code na célula H3
                On Error Resume Next
                wsQR.Pictures("QRCodeImage").Delete ' Remove a imagem anterior se existir
                On Error GoTo 0
                Set img = wsQR.Pictures.Insert(TempFilePath)
                With img
                    .Name = "QRCodeImage"
                    .Top = wsQR.Range("H3").Top
                    .Left = wsQR.Range("H3").Left
                    ' Desativa a proporção fixa e ajusta o tamanho
                    .ShapeRange.LockAspectRatio = msoFalse
                    .Placement = xlMoveAndSize
                    .Width = 11.5 * cmToPoints ' 11,5 cm em pontos
                    .Height = 11.5 * cmToPoints ' 11,5 cm em pontos
                End With
            Else
                MsgBox "Falha ao gerar o QR Code. Verifique sua conexão com a internet.", vbExclamation
                Exit Sub
            End If
            
            ' Imprime a área de impressão "Area_de_impressao"
            wsQR.PageSetup.PrintArea = wsQR.Range("Area_de_impressao").Address
            wsQR.PrintOut
            
            ' Pausa por 6 segundos antes de ir para o próximo número
            Application.Wait Now + TimeValue("00:00:01")
        Next i
    Else
        MsgBox "Por favor, insira um número válido em B11.", vbExclamation
    End If
End Sub
