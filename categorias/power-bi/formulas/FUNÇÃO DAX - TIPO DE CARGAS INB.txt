CARD_INB_BATIDA = 

VAR BATIDA = 
CALCULATE(
SUM(CONS_INB_TP_CARGA[SOMA]),
CONS_INB_TP_CARGA[TIPO_CARGA] = "BATIDA")


VAR A = 
BATIDA&" Cargas Batidas"

RETURN
