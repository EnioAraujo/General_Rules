   Fonte = Sql.Database("192.168.102.154", "logixprd", [
        Query =
        "SELECT
    E.ENDERECO,
    E.DES_ENDERECO,
    E.ITEM,
    E.ITEM_DEPOSIT,
    E.DEN_ITEM,
    E.SKU,
    E.DES_TIP_ESTOQUE,
    E.DES_RESTRICAO,
    E.QTD_REGULARIZADA,
    E.QTD_SALDO_FISICO,
    E.QTD_RESERVA_NORMAL,
    E.QTD_RESERVA_PROVIS,
    E.QTD_SALDO_DISPONIVEL,
    E.PALETE,
    E.CTR_ENT_SAI_VEIC_DOCUM,
    E.NUM_LOTE,

            RTRIM(LTRIM(IDEN_ESTOQUE_AGRUPADA))               AS IDEN_ESTOQUE_AGRUPADA,
            CAST(DAT_HOR_PRODUCAO AS DATETIME)                AS DAT_HOR_PRODUCAO,
            CAST(DAT_HOR_VALIDADE AS DATETIME)                AS DAT_HOR_VALIDADE,
            CAST(DAT_RECEBIMENTO AS DATETIME)                 AS DAT_RECEBIMENTO,

            -- Lógicas matemáticas que antes travavam o Power Query:
