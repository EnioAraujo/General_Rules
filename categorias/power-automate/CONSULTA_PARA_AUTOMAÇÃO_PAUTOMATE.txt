// DAX Query
DEFINE
	VAR __DS0Core = 
		SUMMARIZECOLUMNS(
			ROLLUPADDISSUBTOTAL(
				ROLLUPGROUP(
					'[FAT-BD] SALDO DE ESTOQUE'[DES_TIP_ESTOQUE],
					'[FAT-BD] SALDO DE ESTOQUE'[ENDERECO_2],
					'[FAT-BD] SALDO DE ESTOQUE'[DES_RESTRICAO],
					'[FAT-BD] SALDO DE ESTOQUE'[ITEM_DEPOSIT],
					'[FAT-BD] SALDO DE ESTOQUE'[DEN_ITEM],
					'[FAT-BD] SALDO DE ESTOQUE'[NUM_LOTE],
					'[FAT-BD] SALDO DE ESTOQUE'[DAT_RECEBIMENTO],
					'[FAT-BD] SALDO DE ESTOQUE'[DAT_HOR_VALIDADE],
					'[FAT-BD] SALDO DE ESTOQUE'[IDENTIF_ESTOQUE],
					'[FAT-BD] SALDO DE ESTOQUE'[CTR_ENT_SAI_VEIC_DOCUM]
				), "IsGrandTotalRowTotal"
			),
			"SumQTD_BLOQUEADO", CALCULATE(SUM('[FAT-BD] SALDO DE ESTOQUE'[QTD_BLOQUEADO])),
			"SumQTD_RESERVA_NORMAL", CALCULATE(SUM('[FAT-BD] SALDO DE ESTOQUE'[QTD_RESERVA_NORMAL])),
			"SumQTD_SALDO_FISICO", CALCULATE(SUM('[FAT-BD] SALDO DE ESTOQUE'[QTD_SALDO_FISICO])),
			"SumSALDO_REAL", CALCULATE(SUM('[FAT-BD] SALDO DE ESTOQUE'[SALDO_REAL]))
		)

	VAR __DS0PrimaryWindowed = 
		TOPN(
			40.000,
			__DS0Core,
			[IsGrandTotalRowTotal],
			0,
			'[FAT-BD] SALDO DE ESTOQUE'[DES_TIP_ESTOQUE],
			1,
			'[FAT-BD] SALDO DE ESTOQUE'[ENDERECO_2],
			1,
			'[FAT-BD] SALDO DE ESTOQUE'[DES_RESTRICAO],
			1,
			'[FAT-BD] SALDO DE ESTOQUE'[ITEM_DEPOSIT],
			1,
			'[FAT-BD] SALDO DE ESTOQUE'[DEN_ITEM],
			1,
			'[FAT-BD] SALDO DE ESTOQUE'[NUM_LOTE],
			1,
			'[FAT-BD] SALDO DE ESTOQUE'[DAT_RECEBIMENTO],
			1,
			'[FAT-BD] SALDO DE ESTOQUE'[DAT_HOR_VALIDADE],
			1,
			'[FAT-BD] SALDO DE ESTOQUE'[IDENTIF_ESTOQUE],
			1,
			'[FAT-BD] SALDO DE ESTOQUE'[CTR_ENT_SAI_VEIC_DOCUM],
			1
		)

EVALUATE
	__DS0PrimaryWindowed

ORDER BY
	[IsGrandTotalRowTotal] DESC,
	'[FAT-BD] SALDO DE ESTOQUE'[DES_TIP_ESTOQUE],
	'[FAT-BD] SALDO DE ESTOQUE'[ENDERECO_2],
	'[FAT-BD] SALDO DE ESTOQUE'[DES_RESTRICAO],
